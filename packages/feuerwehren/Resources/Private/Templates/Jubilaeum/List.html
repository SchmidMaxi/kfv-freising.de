document.addEventListener('DOMContentLoaded', () => { const table = document.getElementById('jubilaeum-table'); if (!table) return; const tbody = table.querySelector('tbody'); const dirs = {}; table.querySelectorAll('.jubilaeum-sort').forEach(btn => { btn.addEventListener('click', () => { const col = parseInt(btn.dataset.sortCol, 10); const dir = dirs[col] === 'asc' ? 'desc' : 'asc'; dirs[col] = dir; table.querySelectorAll('.jubilaeum-sort').forEach(b => { b.setAttribute('aria-sort', 'none'); b.querySelector('[data-icon="both"]').classList.remove('d-none'); b.querySelector('[data-icon="both"]').classList.add('opacity-50'); b.querySelector('[data-icon="asc"]').classList.add('d-none'); b.querySelector('[data-icon="desc"]').classList.add('d-none'); }); btn.setAttribute('aria-sort', dir === 'asc' ? 'ascending' : 'descending'); btn.querySelector('[data-icon="both"]').classList.add('d-none'); btn.querySelector('[data-icon="asc"]').classList.toggle('d-none', dir !== 'asc'); btn.querySelector('[data-icon="desc"]').classList.toggle('d-none', dir !== 'desc'); const rows = Array.from(tbody.querySelectorAll('tr')); rows.sort((a, b) => { const aText = (a.querySelectorAll('td')[col]?.textContent ?? '').trim(); const bText = (b.querySelectorAll('td')[col]?.textContent ?? '').trim(); const cmp = aText.localeCompare(bText, 'de', { numeric: true }); return dir === 'asc' ? cmp : -cmp; }); rows.forEach(row => tbody.appendChild(row)); }); }); });
