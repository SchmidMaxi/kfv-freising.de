document.addEventListener('DOMContentLoaded', () => { const root = document.getElementById('termine-kalender-root'); if (!root) return; let events = []; try { events = JSON.parse(root.dataset.events || '[]'); } catch (e) { events = []; } const grid = document.getElementById('termine-cal-grid'); const label = document.getElementById('termine-cal-label'); const monthCount = document.getElementById('termine-cal-month-count'); const monthList = document.getElementById('termine-cal-month-list'); const selectedBox = document.getElementById('termine-cal-selected'); const monthNames = ['Januar', 'Februar', 'März', 'April', 'Mai', 'Juni', 'Juli', 'August', 'September', 'Oktober', 'November', 'Dezember']; const parseDate = (s) => { const [y, m, d] = s.split('-').map(Number); return new Date(y, m - 1, d); }; const isoDate = (d) => `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, '0')}-${String(d.getDate()).padStart(2, '0')}`; const fmtDate = (d) => d.toLocaleDateString('de-DE', { weekday: 'long', day: '2-digit', month: 'long', year: 'numeric' }); const eventsByDate = {}; events.forEach((e) => { (eventsByDate[e.date] = eventsByDate[e.date] || []).push(e); }); let current = events.length > 0 ? Wählen Sie ein Datum im Kalender, um die Details zu sehen.
`; return; } const dayEvents = eventsByDate[selected] || []; const header = document.createElement('div'); header.className = 'gradient-fire rounded-3 p-3 text-white mb-3'; header.innerHTML = `
Ausgewähltes Datum
${fmtDate(parseDate(selected))}
${dayEvents.length} Termin(e)
`; selectedBox.appendChild(header); if (dayEvents.length === 0) { const p = document.createElement('div'); p.className = 'card border-0'; p.innerHTML = '
Keine Termine an diesem Tag.
'; selectedBox.appendChild(p); } else { dayEvents.forEach((e) => selectedBox.appendChild(eventCard(e))); } } function renderMonthList() { const y = current.getFullYear(); const m = current.getMonth(); const inMonth = events.filter((e) => { const d = parseDate(e.date); return d.getFullYear() === y && d.getMonth() === m; }); monthCount.textContent = `${inMonth.length} Termine im ${monthNames[m]}`; monthList.innerHTML = ''; if (inMonth.length === 0) { monthList.innerHTML = '

Keine Termine in diesem Monat.

'; return; } inMonth.forEach((e) => { const btn = document.createElement('button'); btn.type = 'button'; btn.className = 'btn btn-sm btn-light text-start d-flex align-items-center gap-2 w-100'; const d = parseDate(e.date); const dateSpan = document.createElement('span'); dateSpan.className = 'text-body-secondary small'; dateSpan.textContent = `${String(d.getDate()).padStart(2, '0')}.${String(d.getMonth() + 1).padStart(2, '0')}.`; const titleSpan = document.createElement('span'); titleSpan.className = 'text-truncate'; titleSpan.textContent = e.title; btn.append(dateSpan, titleSpan); btn.addEventListener('click', () => { selected = e.date; renderSelected(); }); monthList.appendChild(btn); }); } function renderGrid() { grid.innerHTML = ''; label.textContent = `${monthNames[current.getMonth()]} ${current.getFullYear()}`; const firstOfMonth = new Date(current.getFullYear(), current.getMonth(), 1); const startOffset = (firstOfMonth.getDay() + 6) % 7; const daysInMonth = new Date(current.getFullYear(), current.getMonth() + 1, 0).getDate(); for (let i = 0; i < startOffset; i++) { const empty = document.createElement('div'); grid.appendChild(empty); } for (let day = 1; day <= daysInMonth; day++) { const dateObj = new Date(current.getFullYear(), current.getMonth(), day); const iso = isoDate(dateObj); const hasEvent = !!eventsByDate[iso]; const btn = document.createElement('button'); btn.type = 'button'; btn.className = 'btn btn-sm rounded-circle p-0 d-flex align-items-center justify-content-center position-relative'; btn.style.aspectRatio = '1'; btn.style.width = '100%'; btn.classList.add(iso === selected ?
